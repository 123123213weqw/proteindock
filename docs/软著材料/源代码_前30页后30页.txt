def normalize_element(symbol: str) -> str:
    """把 PDB/SDF 中可能出现的元素写法统一为大写规范形式。

    :param symbol: 原始元素符号, 如 "c", "CA", "Cl"
    :return: 规范化元素符号, 未知元素返回 "C" 并保持兼容

    if sym in ELEMENT_TABLE:
        return sym
    # 尝试首字母匹配(例如 "c1" 之类的 aromatic 标记已在上游剥离)
    if sym[:1] in ELEMENT_TABLE:
        return sym[:1]
    return "C"

def pair_coefficients(type_a: str, type_b: str):
    """返回原子对 (a,b) 的 5 维打分系数。

    规则: gauss/repulsion 取两者均值;
          hydrophobic 仅在双方均为疏水类型时取均值, 否则为 0;
          hbond 仅在双方构成供体-受体对时取均值, 否则为 0。
    """
    ca = TYPE_COEFFICIENTS[type_a]
    cb = TYPE_COEFFICIENTS[type_b]
    g1 = (ca[0] + cb[0]) / 2.0
    g2 = (ca[1] + cb[1]) / 2.0
    rep = (ca[2] + cb[2]) / 2.0
    hydro = (ca[3] + cb[3]) / 2.0 if (ca[3] > 0 and cb[3] > 0) else 0.0
    hb = (ca[4] + cb[4]) / 2.0 if (ca[4] > 0 and cb[4] > 0) else 0.0
    return g1, g2, rep, hydro, hb

# -*- coding: utf-8 -*-
"""刚体几何运算: 四元数、旋转平移变换与 RMSD

 对接搜索中配体的姿态(pose)由「平移向量 + 单位四元数 + 可旋转键扭转角」
 参数化。使用四元数而非欧拉角可以避免万向节锁, 且数值更稳定。
"""

import numpy as np


# ---------------------------------------------------------------- 四元数工具
def quaternion_identity() -> np.ndarray:
    """单位四元数 (w, x, y, z)。"""
[第 5 页]

    return np.array([1.0, 0.0, 0.0, 0.0])


def random_quaternion(rng: np.random.Generator) -> np.ndarray:
    """均匀采样 SO(3) 上的单位四元数 (Shoemake 方法)。"""
    u1, u2, u3 = rng.random(3)
    q = np.array([
        np.sqrt(1.0 - u1) * np.sin(2.0 * np.pi * u2),
        np.sqrt(1.0 - u1) * np.cos(2.0 * np.pi * u2),
        np.sqrt(u1) * np.sin(2.0 * np.pi * u3),
        np.sqrt(u1) * np.cos(2.0 * np.pi * u3),
    ])
    return q / np.linalg.norm(q)


def quaternion_normalize(q: np.ndarray) -> np.ndarray:
    """把四元数归一化为单位四元数。"""
    n = np.linalg.norm(q)
    if n < 1e-12:
        return quaternion_identity()
    return q / n


def quaternion_multiply(q1: np.ndarray, q2: np.ndarray) -> np.ndarray:
    """四元数乘法 q1 * q2。"""
    w1, x1, y1, z1 = q1
    w2, x2, y2, z2 = q2
    return np.array([
        w1 * w2 - x1 * x2 - y1 * y2 - z1 * z2,
        w1 * x2 + x1 * w2 + y1 * z2 - z1 * y2,
        w1 * y2 - x1 * z2 + y1 * w2 + z1 * x2,
        w1 * z2 + x1 * y2 - y1 * x2 + z1 * w2,
    ])


def quaternion_rotate(q: np.ndarray, v: np.ndarray) -> np.ndarray:
    """用四元数旋转向量或向量组。

    :param q: 单位四元数 (w,x,y,z)
    :param v: (3,) 或 (N,3) 坐标数组
    :return: 旋转后同形状坐标数组
    """
    v = np.asarray(v, dtype=np.float64)
    single = v.ndim == 1
    pts = np.atleast_2d(v)
    w, x, y, z = q
    # 旋转矩阵
    rot = np.array([
        [1 - 2 * (y * y + z * z), 2 * (x * y - w * z), 2 * (x * z + w * y)],
        [2 * (x * y + w * z), 1 - 2 * (x * x + z * z), 2 * (y * z - w * x)],
[第 6 页]

        [2 * (x * z - w * y), 2 * (y * z + w * x), 1 - 2 * (x * x + y * y)],
    ])
    out = pts @ rot.T
    return out[0] if single else out


def quaternion_to_matrix(q: np.ndarray) -> np.ndarray:
    """四元数转 3x3 旋转矩阵。"""
    return quaternion_rotate(q, np.eye(3)).T


def small_random_quaternion(rng: np.random.Generator, max_angle_rad: float) -> np.ndarray:
    """在当前姿态附近采样一个随机小旋转(轴均匀, 角度上限给定)。"""
    axis = rng.normal(size=3)
    axis /= max(np.linalg.norm(axis), 1e-12)
    angle = rng.uniform(0.0, max_angle_rad)
    half = angle / 2.0
    q = np.array([np.cos(half), *(np.sin(half) * axis)])
    return quaternion_normalize(q)


# ---------------------------------------------------------------- 刚体变换
class RigidTransform:
    """平移 + 旋转构成的刚体变换, 作用于质心坐标系最为方便。"""

    __slots__ = ("translation", "rotation_q")

    def __init__(self, translation=None, rotation_q=None):
        self.translation = np.zeros(3) if translation is None else np.asarray(translation, float)
        self.rotation_q = quaternion_identity() if rotation_q is None else np.asarray(rotation_q, float)

    def apply(self, coords: np.ndarray) -> np.ndarray:
        """对以质心为原点的坐标组施加变换。"""
        return quaternion_rotate(self.rotation_q, coords) + self.translation

    def compose(self, d_translation: np.ndarray, d_rotation: np.ndarray) -> None:
        """在当前变换上叠加一个小扰动(先旋转后平移)。"""
        self.rotation_q = quaternion_normalize(
            quaternion_multiply(d_rotation, self.rotation_q))
        self.translation = self.translation + d_translation


def centered(coords: np.ndarray) -> np.ndarray:
    """返回以质心为原点的坐标副本。"""
    return coords - coords.mean(axis=0)


# ---------------------------------------------------------------- RMSD
def rmsd(coords_a: np.ndarray, coords_b: np.ndarray) -> float:
    """直接 RMSD(未做叠合), 用于对接构象与晶体构象比较。"""
[第 7 页]

    diff = np.asarray(coords_a) - np.asarray(coords_b)
    return float(np.sqrt((diff ** 2).sum() / max(len(diff), 1)))


def kabsch_rmsd(P: np.ndarray, Q: np.ndarray) -> float:
    """Kabsch 叠合后最小 RMSD, 忽略整体朝向差异。"""
    Pc = centered(P)
    Qc = centered(Q)
    H = Pc.T @ Qc
    U, _, Vt = np.linalg.svd(H)
    d = np.sign(np.linalg.det(Vt.T @ U.T))
    D = np.diag([1.0, 1.0, d])
    R = Vt.T @ D @ U.T
    prot = Pc @ R.T
    return rmsd(prot, Qc)

@dataclass
class Protein:
    """蛋白质结构对象(重原子表示)。"""

    coords: np.ndarray          # (N,3) 原子坐标
    types: list                 # (N,) 打分类型字符串

    def n_atoms(self) -> int:

    def centroid(self) -> np.ndarray:

    def pocket_atoms_mask(self, center: np.ndarray, radius: float = 10.0) -> np.ndarray:
        """返回距口袋中心 radius 以内的原子布尔掩码。"""
        d2 = ((self.coords - np.asarray(center)) ** 2).sum(axis=1)
        return d2 <= radius * radius

    def write_pdb(self, path: str) -> None:
        """把蛋白质(通常为加氢前后处理版本)写回 PDB。"""
        with open(path, "w", encoding="utf-8") as fh:
            for i in range(self.n_atoms):
                x, y, z = self.coords[i]
                fh.write(
                    "ATOM  {:5d} {:>4s} {:3s} {:1s}{:4d}    "
                    "{:8.3f}{:8.3f}{:8.3f}  1.00  0.00          {:>2s}\n".format(
                        i + 1, self.atom_names[i], self.res_names[i],
                        self.chains[i], int(self.res_ids[i]), x, y, z,
                        self.elements[i],
                    )
                )
            fh.write("TER\nEND\n")


def parse_pdb(path: str, keep_hetero: bool = False) -> Protein:
    """解析 PDB 文件, 提取标准氨基酸重原子(可选保留 HETATM 中的非水杂原子)。

    :param path: PDB 文件路径
    :param keep_hetero: 是否保留金属等 HETATM 记录(忽略水与常见辅因子)
    :return: Protein 对象
    """
    coords, types, elements = [], [], []
    atom_names, res_names, res_ids, chains = [], [], [], []
    skip_res = {"HOH", "WAT", "DOD", "SOL", "HEM", "NAG", "MAN", "SO4", "PO4", "GOL"}

    with open(path, encoding="utf-8", errors="ignore") as fh:
        for line in fh:
            rec = line[:6].strip()
            if rec not in ("ATOM", "HETATM"):
                continue
            res = line[17:20].strip().upper()
[第 9 页]

            if rec == "HETATM":
                if not keep_hetero or res in skip_res:
                    continue
            elif res not in STANDARD_RESIDUES:
                continue
            name = line[12:16].strip()
            element = line[76:78].strip()
            if not element:
                # 依据原子名推断元素
                element = "".join(ch for ch in name if ch.isalpha())[:1]
            element = normalize_element(element)
            if element == "H":
                continue
            x = float(line[30:38]); y = float(line[38:46]); z = float(line[46:54])
            chain = line[21].strip() or "A"
            try:
                res_id = int(line[22:26])
            except ValueError:
                res_id = 0
            coords.append([x, y, z])
            types.append(assign_protein_type(element, name, res))
            elements.append(element)
            atom_names.append(name)
            res_names.append(res)
            res_ids.append(res_id)
            chains.append(chain)

    if not coords:
        raise ValueError("PDB 文件中未找到可用原子: {}".format(path))

    return Protein(
        coords=np.array(coords, dtype=np.float64),
        types=types,
        type_ids=np.array([TYPE_INDEX[t] for t in types], dtype=np.int64),
        elements=elements,
        atom_names=atom_names,
        res_names=res_names,
        res_ids=res_ids,
        chains=chains,
        source=path,
    )

def detect_pocket(protein: Protein, spacing: float = 1.0,

    """LigSite 风格的口袋检测(盲对接)。

    原理(Hendlich et al., 1997):
      1. 把蛋白包围盒体素化为 occupied(被重原子+探针半径覆盖)/free;
[第 10 页]

      2. 沿 ±x/±y/±z 六个方向扫描, 一个 free 体素若在某方向上
         一侧遇到蛋白、另一侧通向外部溶剂, 则该方向记为" pocket ";

          平坦表面几乎不被夹住);

    :return: 口袋中心坐标 (3,)
    """
    coords = protein.coords
    pad = 2.5
    lo = coords.min(axis=0) - pad
    hi = coords.max(axis=0) + pad
    n_cells = np.ceil((hi - lo) / spacing).astype(int) + 1
    if float(np.prod(n_cells)) > 6e6:            # 超大蛋白保护
        spacing = float(np.cbrt(np.prod(hi - lo) / 2e6))
        n_cells = np.ceil((hi - lo) / spacing).astype(int) + 1

    from scipy.spatial import cKDTree
    tree = cKDTree(coords)

    gx, gy, gz = np.meshgrid(*axis_pts, indexing="ij")
    vox = np.stack([gx.ravel(), gy.ravel(), gz.ravel()], axis=1)
    nearest, _ = tree.query(vox, k=1)
    mean_vdw = float(np.mean([vdw_radius(e) for e in protein.elements]))
    occ = (nearest <= mean_vdw + probe_radius).reshape(tuple(n_cells))
    free = ~occ

    # 六方向 LigSite 扫描: free 体素在某方向上两侧均被蛋白夹住
    pocket_votes = np.zeros_like(free, dtype=np.int8)
    for axis in range(3):
        cum = np.cumsum(occ, axis=axis)
        tot_idx = [slice(None)] * 3
        tot_idx[axis] = slice(-1, None)
        total = cum[tuple(tot_idx)]              # 该轴占用总数(axis 维为 1)
        before = cum - occ                       # 该点之前的占用数
        after = total - cum                      # 该点之后的占用数
        votes = free & (before > 0) & (after > 0)
        pocket_votes += votes.astype(np.int8)    # 正方向已覆盖双向语义

    candidates = pocket_votes >= min_dirs
    if not candidates.any():
        candidates = pocket_votes >= 2
        if not candidates.any():
            return protein.centroid()
    # 取最高得分子簇: 以得分最高体素为中心半径 4 A 内加权质心
    best_flat = np.argmax(pocket_votes * candidates)
    bi, bj, bk = np.unravel_index(best_flat, pocket_votes.shape)
    center_voxel = np.array([axis_pts[0][bi], axis_pts[1][bj],

                             axis_pts[2][bk]])
    # 邻域 4A 内候选点按票数加权
    lo_i = np.stack(np.meshgrid(
        *[np.arange(n_cells[i]) for i in range(3)], indexing="ij"), axis=-1)
    d2 = ((lo_i * spacing + lo - center_voxel) ** 2).sum(axis=-1)
    near = candidates & (d2 <= 16.0)
    if not near.any():
        return center_voxel
    weights = pocket_votes[near].astype(float)
    pts_voxel = lo_i[near] * spacing + lo

class Ligand:

    :ivar elements: (N,) 元素符号

    :ivar type_ids: (N,) 类型索引
    :ivar rotatable: list[(i, j, movable_atom_ids)] 可旋转键
    :ivar torsions: np.ndarray 当前扭转角(rad), 长度与 rotatable 一致
    """

    def __init__(self, coords, elements, types, rotatable, mol=None, name="ligand"):
        self.coords = np.asarray(coords, dtype=np.float64)
        self.elements = list(elements)
        self.types = list(types)
        self.type_ids = np.array([TYPE_INDEX[t] for t in types], dtype=np.int64)
        self.rotatable = rotatable        # [(begin, end, [movable...])]
        self.torsions = np.zeros(len(rotatable))

        self.name = name
        self._mol = mol                   # 保留 RDKit 分子用于输出

    @property
    def n_atoms(self) -> int:
        return len(self.coords)

    @property
    def n_torsions(self) -> int:
        return len(self.rotatable)

    def copy(self) -> "Ligand":
        lig = Ligand(self.coords.copy(), list(self.elements), list(self.types),
                     self.rotatable, self._mol, self.name)
        lig.torsions = self.torsions.copy()
        return lig

    # ------------------------------------------------------------ 几何
    def conformer(self, transform=None) -> np.ndarray:
        """按当前扭转角重建构象, 可再叠加刚体变换。

        :param transform: chemistry.RigidTransform 或 None
        :return: (N,3) 世界坐标
        """
        coords = self.coords.copy()
        for (b, e, movable), angle in zip(self.rotatable, self.torsions):
            axis = coords[e] - coords[b]
            norm = np.linalg.norm(axis)
            if norm < 1e-6:
                continue
            axis = axis / norm
            c, s = np.cos(angle), np.sin(angle)
            # Rodrigues 旋转公式
            kx, ky, kz = axis
            K = np.array([[0, -kz, ky], [kz, 0, -kx], [-ky, kx, 0]])
            R = np.eye(3) * c + s * K + (1 - c) * np.outer(axis, axis)
            coords[movable] = (coords[movable] - coords[e]) @ R.T + coords[e]
        if transform is not None:
            coords = transform.apply(coords)
        return coords

def build_ligand(mol, name: str = "ligand", keep_origin: bool = False) -> Ligand:
    """从 RDKit Mol 构造 Ligand(自动补 3D、分型、找可旋转键)。

    :param keep_origin: True 保留原始晶格坐标(数据集构建用);
                        False 平移到质心(对接搜索用, 默认)
    """
    mol = Chem.Mol(mol)
    if any(a.GetSymbol() == "H" for a in mol.GetAtoms()):
        mol = Chem.RemoveHs(mol)
    if mol.GetNumConformers() == 0 or mol.GetNumAtoms() < 1:
        mol = Chem.AddHs(mol)
        params = AllChem.ETKDGv3()
        params.randomSeed = ETKDG_SEED
        AllChem.EmbedMolecule(mol, params)
        if mol.GetNumConformers() == 0:
            raise ValueError("无法为配体生成 3D 构象: {}".format(name))
        mol = Chem.RemoveHs(mol)

    conf = mol.GetConformer()
    coords = np.array([list(conf.GetAtomPosition(i)) for i in range(mol.GetNumAtoms())])
    elements = [a.GetSymbol().upper() for a in mol.GetAtoms()]
    types = [_assign_ligand_type(mol, i) for i in range(mol.GetNumAtoms())]
    rotatable = _find_rotatable_bonds(mol)
    if not keep_origin:
        coords = coords - coords.mean(axis=0)
    lig = Ligand(coords, elements, types, rotatable,
                 mol=mol, name=name)
    _refresh_torsions(lig)
    return lig

def _find_rotatable_bonds(mol):

def _movable_side(mol, b, e):
    """把键 (b,e) 断开后, e 一侧的原子集合。"""
    n = mol.GetNumAtoms()
    adjacency = [[] for _ in range(n)]
    for bond in mol.GetBonds():
        i, j = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
        if (i, j) == (b, e) or (j, i) == (b, e):
            continue
        adjacency[i].append(j)
        adjacency[j].append(i)
    seen = {e}
    stack = [e]
    while stack:
        cur = stack.pop()
        for nxt in adjacency[cur]:
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    movable = sorted(seen - {e})
    return [e] + movable if movable else []

    lig.torsions = np.zeros(lig.n_torsions)

def _build_coefficient_tensor():
    """展开 (16,16,5) 原子对系数张量(模块级缓存)。"""
    tensor = np.zeros((N_TYPES, N_TYPES, 5))

            tensor[i, j] = pair_coefficients(a, b)

def _linear_step(values, x_low, x_high):
    """[x_low, x_high] 内 1->0 线性衰减的阶跃基元。"""
    out = np.zeros_like(values)
    if x_high > x_low:
        out[values < x_low] = 1.0
        band = (values >= x_low) & (values <= x_high)
        out[band] = (x_high - values[band]) / (x_high - x_low)
    else:
        out[values < x_high] = 1.0
    return out


def pairwise_term_sum(lig_type_ids, rec_type_ids, d_matrix, cutoff=None):
    """计算两组原子间五项能量之和(全向量化)。

    :param lig_type_ids: (N,) 配体类型索引
    :param rec_type_ids: (M,) 蛋白类型索引
    :param d_matrix: (N,M) 原子间距离矩阵 / A
    :param cutoff: 距离截断, 默认 8 A
    :return: (5,) 各项加权和向量 [gauss1, gauss2, repulsion, hydrophobic, hbond]
    """

    lig_ids = np.asarray(lig_type_ids)
    rec_ids = np.asarray(rec_type_ids)
    d = np.asarray(d_matrix, dtype=np.float64)

    close = d < cutoff
    if not close.any():
        return np.zeros(5)

    rows, cols = np.nonzero(close)
    dist = d[rows, cols]
    coeff = _COEFF_TENSOR[lig_ids[rows], rec_ids[cols]]        # (K,5)
    r_pair = _VDW_VECTOR[lig_ids[rows]] + _VDW_VECTOR[rec_ids[cols]]
    # 表面距离(Vina 平滑距离): ds = d - ri - rj

    g1 = np.exp(-(dist / 0.5) ** 2)
    g2 = np.exp(-((dist - 3.0) / 2.0) ** 2)

    rep_cut = 0.9 * r_pair

                   (rep_cut / np.maximum(dist, 1e-6) - 1.0) ** 2, 0.0)
    hydro = _linear_step(ds, 0.5, 2.0)
    hb = _linear_step(ds, -0.7, 0.0)

    primitives = np.stack([g1, g2, rep, hydro, hb], axis=1)     # (K,5)
    weighted = ((coeff * primitives) * _WEIGHTS).sum(axis=0)    # (5,)
    return weighted


def intermolecular_energy(lig_types_ids, rec_types_ids, d_matrix):
    """蛋白-配体相互作用总能量及分解。

    :return: (total_kcal, terms_dict)

    weighted = pairwise_term_sum(lig_types_ids, rec_types_ids, d_matrix)
    terms = {
        "gauss1": float(weighted[0]),
        "gauss2": float(weighted[1]),
        "repulsion": float(weighted[2]),
        "hydrophobic": float(weighted[3]),
        "hbond": float(weighted[4]),
    }
    return float(weighted.sum()), terms

def _linear_step_batch(values, x_low, x_high):
    """_linear_step 的批量版, values 形状 (..., K) 对 K 种类型。"""
    out = np.zeros_like(values)
    if x_high > x_low:
        out[values < x_low] = 1.0
        band = (values >= x_low) & (values <= x_high)
        out[band] = (x_high - values[band]) / (x_high - x_low)
    else:
        out[values < x_high] = 1.0
    return out

    @classmethod
    def random(cls, ligand, center, box_size, rng):
        """在口袋中心附近随机初始化 pose(高斯采样, 提高落入率)。"""
        sigma = box_size / 6.0

    def perturb(self, ligand, rng, temperature=1.0):
        """温度自适应随机扰动, 返回新 Pose。"""
        scale = max(temperature, 0.05)
        dt = rng.uniform(-1, 1, size=3) * MAX_TRANSLATE * scale
        dq = small_random_quaternion(rng, MAX_ROTATE * scale)
        from ..chemistry.geometry import quaternion_multiply
        new = Pose(
            self.translation + dt,
            quaternion_multiply(dq, self.rotation_q),
            self.torsions.copy(),
        )
        if ligand.n_torsions:
            idx = rng.integers(0, ligand.n_torsions)
            new.torsions[idx] += rng.uniform(-1, 1) * MAX_TORSION * scale
        return new

        tr = RigidTransform(self.translation, self.rotation_q)
        return ligand.conformer(tr)

        return Pose(self.translation.copy(), self.rotation_q.copy(),

def _metropolis(e_old, e_new, temperature, rng):
    """Metropolis 接受准则(能量越低越优)。"""
    if e_new <= e_old:
        return True
    return rng.random() < np.exp(-(e_new - e_old) / max(temperature, 1e-6))


def simulated_annealing(ligand, scorer, center, box_size, rng,
                        n_cycles=8, steps_per_cycle=260,
                        t_init=2.5, t_final=0.02, callback=None):
    """执行一次完整的 MC-SA 搜索。
[第 26 页]


    :param ligand: io.ligand.Ligand
    :param scorer: scoring.hybrid.HybridScorer (使用其 vina 快速通道)
    :param center: 口袋中心
    :param box_size: 搜索盒子边长 / A
    :param rng: numpy 随机生成器
    :return: dict(pose, score, trajectory)
    """
    pose = Pose.random(ligand, center, box_size, rng)
    energy = scorer.vina_score(ligand, pose.coords(ligand))["total"]
    best_pose, best_energy = pose.copy(), energy
    trajectory = [energy]

    total_steps = n_cycles * steps_per_cycle
    for step in range(total_steps):
        frac = step / max(total_steps - 1, 1)
        # 几何降温 + 周期性回火
        cycle_frac = (step % steps_per_cycle) / steps_per_cycle
        reheat = 1.0 + 0.6 * np.cos(2 * np.pi * (step // steps_per_cycle))
        temperature = (t_init * (t_final / t_init) ** frac) * reheat \
            if cycle_frac > 0 else t_init

        cand = pose.perturb(ligand, rng, temperature)
        cand_energy = scorer.vina_score(ligand, cand.coords(ligand))["total"]
        if _metropolis(energy, cand_energy, temperature, rng):
            pose, energy = cand, cand_energy
            if energy < best_energy:
                best_energy = energy
                best_pose = pose.copy()
        trajectory.append(energy)
        if callback is not None:
            callback(step, total_steps, energy, best_energy)

    return {"pose": best_pose, "score": best_energy, "trajectory": trajectory}

        # 旋转差异: 四元数内积
        dot = abs(float(np.dot(pose_a.rotation_q, pose_b.rotation_q)))
        ang = 2.0 * np.arccos(min(1.0, dot))
        # 经验映射: 平移 + 旋转角(rad)*1.5A 近似 RMSD
        return d_t + 1.5 * ang

    # ------------------------------------------------------------ 输出
    def save(self, save_dir):
        """输出 top 构象 SDF 与 report.json / report 文本。"""
        os.makedirs(save_dir, exist_ok=True)
        for res in self.results:
            out_sdf = os.path.join(save_dir,
                                   "pose_{}.sdf".format(res.rank))
            write_sdf(self.ligand, res.coords, out_sdf,
                      score=res.scores.get("final"), 
                      name="{}_rank{}".format(self.ligand.name, res.rank))
        # 蛋白-配体复合物 PDB(取最优构象)
        if self.results:
            best = self.results[0]
            complex_path = os.path.join(save_dir, "complex_best.pdb")
            with open(complex_path, "w", encoding="utf-8") as fh:
                for i in range(self.protein.n_atoms):
                    x, y, z = self.protein.coords[i]
                    fh.write(
                        "ATOM  {:5d} {:>4s} {:3s} {:1s}{:4d}    "

function pollStatus() {
  App.pollTimer = setInterval(() => {
    fetch("/api/jobs/" + App.jobId).then(r => r.json()).then(job => {
      setProgress(job.progress || 0, job.stage, job.message);
      if (job.status === "done") {
        clearInterval(App.pollTimer);
        onJobDone(job);
      } else if (job.status === "error") {
        clearInterval(App.pollTimer);
        setProgress(100, "error", "失败: " + job.error);
      }
    }).catch(() => {});
  }, 1500);
}

function drawBars(ctx, items, canvas) {
  const w = canvas.width, h = canvas.height, pad = 34;
  const vmin = Math.min(0, ...items.map(i => i[1]));
  const vmax = Math.max(0.1, ...items.map(i => i[1]));
  const y0 = h - pad;
  const scale = (h - pad * 2) / (vmax - vmin);
  ctx.strokeStyle = "#3a4560";
  ctx.beginPath(); ctx.moveTo(pad, y0 - (0 - vmin) * scale);
  ctx.lineTo(w - 8, y0 - (0 - vmin) * scale); ctx.stroke();
  const bw = (w - pad * 2) / items.length * 0.55;
  items.forEach((it, i) => {
    const cx = pad + (w - pad * 2) * (i + 0.5) / items.length;
    const val = it[1];
    const yTop = y0 - (Math.max(val, 0) - vmin) * scale;
    const yBot = y0 - (Math.min(val, 0) - vmin) * scale;
    ctx.fillStyle = it[2];
    ctx.fillRect(cx - bw / 2, yTop, bw, Math.max(yBot - yTop, 2));
    ctx.fillStyle = "#8b93a7";
    ctx.font = "11px sans-serif"; ctx.textAlign = "center";
    ctx.fillText(it[0], cx, h - 12);
    ctx.fillText(val.toFixed(2), cx, yTop - 6);
  });
}

function drawLossCurve(history) {
  const canvas = $("#loss-canvas");

  const ctx = canvas.getContext("2d");
  const w = canvas.width, h = canvas.height, pad = 30;

  const all = history.train.concat(history.val);
  const vmax = Math.max(...all), vmin = Math.min(...all);
  const x = (i, n) => pad + (w - pad * 1.4) * i / Math.max(n - 1, 1);
  const y = (v) => h - pad - (h - pad * 1.8) * (v - vmin) / Math.max(vmax - vmin, 1e-6);
  [["train", "#4f8cff"], ["val", "#22c1a4"]].forEach(([key, color]) => {

def download_pdb(pdb_id: str, dest: str) -> bool:
    """从 RCSB 下载 PDB 文件。"""
    url = RCSB_URL.format(pdb_id.upper())
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "ProteinDock/1.0"})
        with urllib.request.urlopen(req, timeout=60) as resp:
            data = resp.read()
    except Exception as exc:
        print("下载失败 {}: {}".format(pdb_id, exc))
        return False
    with open(dest, "wb") as fh:
        fh.write(data)
    return True


def split_complex(pdb_path: str, min_lig_atoms: int = 5):
    """把复合物 PDB 拆为蛋白文本与晶体配体 HETATM 块。

    选择规则: 收集全部非水/非辅基 HETATM 残基,
    取原子数不少于 min_lig_atoms 的第一个(通常是主配体)。
    """
    protein_lines, het_by_res = [], {}
    skip = {"HOH", "WAT", "SO4", "GOL", "EDO", "NAG", "MAN", "CL",
            "NA", "MG", "ZN", "CA", "K", "ACT", "MES", "TRS", "PEG",
            "DMS", "EOH", "IPA", "FMT", "ACE"}
    with open(pdb_path, encoding="utf-8", errors="ignore") as fh:
        for line in fh:
            rec = line[:6].strip()
            if rec == "ATOM":
                protein_lines.append(line)
            elif rec == "HETATM":
                res = line[17:20].strip().upper()
                if res in skip:
                    continue
                het_by_res.setdefault(res, []).append(line)
    lig_res, lig_lines = None, []
    for res, lines in het_by_res.items():
        if len(lines) >= min_lig_atoms:

            lig_res, lig_lines = res, lines
            break
    if lig_res is None and het_by_res:      # 回退: 原子最多的 HETATM
        lig_res = max(het_by_res, key=lambda r: len(het_by_res[r]))
        lig_lines = het_by_res[lig_res]
    lig_pdb = "".join(lig_lines) + "END\n"
    return "".join(protein_lines) + "TER\nEND\n", lig_pdb, lig_res

def build_systems(out_root: str, n_decoys: int, max_atoms=70):
    """逐体系构造 dockdata 目录。"""
    from rdkit import Chem
    os.makedirs(out_root, exist_ok=True)
    tmp_dir = os.path.join(out_root, "_tmp")
    os.makedirs(tmp_dir, exist_ok=True)
    built = 0
    for pdb_id in PDB_IDS:
        sys_dir = os.path.join(out_root, pdb_id.lower())
        if os.path.isdir(sys_dir) and os.path.exists(
                os.path.join(sys_dir, "labels.npy")):
            print("跳过已存在: {}".format(sys_dir))
            built += 1
            continue
        raw = os.path.join(tmp_dir, pdb_id + ".pdb")
        if not os.path.exists(raw) and not download_pdb(pdb_id, raw):
            continue
        try:
            protein_text, lig_pdb, lig_res = split_complex(raw)
            if not lig_pdb.strip() or lig_res is None:
                print("{} 无配体, 跳过".format(pdb_id))
                continue
            write_tmp(protein_text, os.path.join(tmp_dir, "p.pdb"))
            write_tmp(lig_pdb, os.path.join(tmp_dir, "l.pdb"))
            protein = parse_pdb(os.path.join(tmp_dir, "p.pdb"))
            lig_mol = Chem.MolFromPDBFile(os.path.join(tmp_dir, "l.pdb"),
                                          removeHs=True, sanitize=True,
                                          proximityBonding=True)
            if lig_mol is None or lig_mol.GetNumAtoms() < 5 \
                    or lig_mol.GetNumAtoms() > max_atoms:
                print("{} 配体无效/过大({})".format(
                    pdb_id, lig_mol.GetNumAtoms() if lig_mol else 0))
                continue
            # 原位配体(保留晶体坐标, native pose / 标签用)
            ligand = build_ligand(lig_mol, name=pdb_id, keep_origin=True)
            # 居中副本(刚体 Pose 采样假设质心坐标系)
[第 56 页]

            ligand_c = build_ligand(lig_mol, name=pdb_id)
            native = ligand.coords + 0.0
            center = native.mean(axis=0)

            # decoys: native 附近小扰动 + 远处随机 pose
            rng = np.random.default_rng(42)
            from proteindock.chemistry.geometry import quaternion_rotate
            lig_center = native.mean(axis=0)
            poses = [native]
            for k in range(n_decoys):
                if k < n_decoys // 3:   # near-native 扰动(绕质心小幅刚体扰动)
                    q = random_quaternion(rng)
                    moved = quaternion_rotate(q, native - lig_center) + lig_center
                    moved += rng.normal(0, 0.6, size=3)
                    poses.append(moved)
                else:                    # 远处 pose: 刚体打散 + 扭转随机
                    pose = Pose.random(ligand_c, center, 16.0, rng)
                    poses.append(pose.coords(ligand_c))

            labels = []
            for coords in poses:
                d = np.sqrt(((coords[:, None, :] - protein.coords[None, :, :]) ** 2)
                            .sum(axis=2))
                pocket = d.min(axis=1) < 12.0
                if not pocket.any():     # 远离蛋白的 pose 给平庸标签
                    labels.append(0.0)
                    continue
                total, _ = intermolecular_energy(
                    ligand_c.type_ids, protein.type_ids, d)
                labels.append(float(total))
            labels = np.array(labels, dtype=np.float32)

            os.makedirs(sys_dir, exist_ok=True)
            with open(os.path.join(sys_dir, "protein.pdb"), "w",
                      encoding="utf-8") as fh:
                fh.write(protein_text)
            from proteindock.io.ligand import write_sdf
            write_sdf(ligand, native, os.path.join(sys_dir, "ligand.sdf"),
                      name=pdb_id + "_native")
            np.save(os.path.join(sys_dir, "decoys.npy"),
                    np.array(poses, dtype=np.float32))
            np.save(os.path.join(sys_dir, "labels.npy"), labels)
            print("构建完成 {} (decoys={}, 标签范围 {:.1f}~{:.1f})".format(
                pdb_id, len(poses), labels.min(), labels.max()))
            built += 1
        except Exception as exc:
            print("{} 处理失败: {}".format(pdb_id, exc))
    print("共构建 {} 个体系 -> {}".format(built, out_root))
    return built
